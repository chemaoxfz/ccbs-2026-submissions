Week 4 — Extension: When slow sampling hides promoter switches

Author: Zheng Haobo · Week 4 (2026-09-28) · CCBS 2026

One sentenceWith equal switching hazard k, endpoint-only samples every Δ minutes miss even numbers of jumps, so pΔ/Δ underestimates k; the model-based inverse is finite only when p^Δ<12.

The point I am adding

Let a promoter be OFF or ON, with constant switch hazard k in either direction: over a short dt, switching has probability kdt+o(dt). Starting OFF, let q(t) be the probability of ON. Gains minus losses give q′=k(1−q)−kq, with q(0)=0; solving at t=Δ gives:

pΔ=1−e−2kΔ2

By symmetry this is the endpoint-mismatch probability from either starting state. A mismatch means an odd number of jumps; an even number is invisible. Counting each mismatch as one event gives naive rate pΔ/Δ, below k because 1−e−x<x for x>0. The public Lecture 7 exposition supports the constant-hazard, continuous-time Markov setup.[1]

Work it

Toy assumptions, not measurements: k=1 min−1, Δ∈{0.1,1,5} min. With seed 20260922, the code generates 200 independent CTMC records per interval, each with 4,000 endpoint samples, using exponential dwell times.

For a record, p^Δ is its mismatch fraction. The correction is:

k^=−ln(1−2p^Δ)2Δ only if p^Δ<12

At equality, k^→∞; above it, no finite inverse exists. Neither is clipped or dropped. At 5 min, 90/200 records (45.0%) reached or exceeded one half: 3 at the boundary, 87 above; 110 had finite corrections.

#!/usr/bin/env python3
"""Simulate a symmetric two-state CTMC and plot endpoint-sampling aliasing.

All rates and sampling intervals are illustrative assumptions, not measurements.
The simulation follows exponential CTMC dwell times, samples only endpoint states,
and retains every replicate's finite/boundary/incompatible inversion status.
"""

import argparse
import json
import math
import random

K_PER_MIN = 1.0
DELTAS_MIN = (0.1, 1.0, 5.0)
SEED = 20260922
N_REPLICATES = 200
N_INTERVALS = 4000


def exponential_wait(rng, rate):
    """Draw an exponential dwell time for a constant transition hazard."""
    return -math.log1p(-rng.random()) / rate


def exact_mismatch_probability(rate, delta):
    """Return (1 - exp(-2 rate delta))/2, evaluated stably near zero."""
    return -0.5 * math.expm1(-2.0 * rate * delta)


def corrected_rate(p_hat, delta):
    """Invert the symmetric CTMC endpoint probability only on its finite domain."""
    return -math.log1p(-2.0 * p_hat) / (2.0 * delta)


def simulate_dataset(rate, delta, n_intervals, rng):
    """Generate one continuous-time path and return endpoint mismatch frequency."""
    state = rng.randrange(2)
    previous_sample = state
    next_jump = exponential_wait(rng, rate)
    mismatches = 0

    for interval_index in range(n_intervals):
        sample_time = (interval_index + 1) * delta
        while next_jump <= sample_time:
            state = 1 - state
            next_jump += exponential_wait(rng, rate)
        if state != previous_sample:
            mismatches += 1
        previous_sample = state

    p_hat = mismatches / n_intervals
    naive_rate = p_hat / delta
    if 2 * mismatches < n_intervals:
        inversion_status = "finite"
        k_hat = corrected_rate(p_hat, delta)
    elif 2 * mismatches == n_intervals:
        inversion_status = "boundary_infinite"
        k_hat = None
    else:
        inversion_status = "above_half_no_finite_inverse"
        k_hat = None

    return {
        "mismatches": mismatches,
        "n_intervals": n_intervals,
        "p_hat": p_hat,
        "naive_rate_per_min": naive_rate,
        "k_hat_per_min": k_hat,
        "inversion_status": inversion_status,
    }


def quantiles(values):
    """Linear-interpolated empirical 10th, 50th, and 90th percentiles."""
    ordered = sorted(values)
    if not ordered:
        return None

    def at(probability):
        position = (len(ordered) - 1) * probability
        low = int(math.floor(position))
        high = int(math.ceil(position))
        weight = position - low
        return ordered[low] * (1.0 - weight) + ordered[high] * weight

    return {"p10": at(0.10), "p50": at(0.50), "p90": at(0.90)}


def summarize_delta(rate, delta, records):
    """Summarize all replicates; finite-only quantiles keep their denominator."""
    counts = {
        "finite": sum(r["inversion_status"] == "finite" for r in records),
        "boundary_infinite": sum(r["inversion_status"] == "boundary_infinite" for r in records),
        "above_half_no_finite_inverse": sum(
            r["inversion_status"] == "above_half_no_finite_inverse" for r in records
        ),
    }
    n_replicates = len(records)
    n_nonfinite = counts["boundary_infinite"] + counts["above_half_no_finite_inverse"]
    finite_estimates = [r["k_hat_per_min"] for r in records if r["inversion_status"] == "finite"]
    return {
        "delta_min": delta,
        "exact_p_delta": exact_mismatch_probability(rate, delta),
        "exact_naive_rate_per_min": exact_mismatch_probability(rate, delta) / delta,
        "p_hat_range_p10_p50_p90": quantiles([r["p_hat"] for r in records]),
        "naive_rate_range_p10_p50_p90": quantiles([r["naive_rate_per_min"] for r in records]),
        "corrected_finite_only_range_p10_p50_p90": quantiles(finite_estimates),
        "corrected_finite_only_n": len(finite_estimates),
        "n_replicates": n_replicates,
        "status_counts": counts,
        "p_hat_ge_half_fraction": n_nonfinite / n_replicates,
    }


def _fmt(value, digits=3):
    return "—" if value is None else f"{value:.{digits}f}"


def make_svg(summaries, rate):
    """Build the figure as inline SVG using page-level CSS custom properties."""
    width, height = 920, 390
    x_left, x_right = 125.0, 820.0
    y_top, y_bottom = 34.0, 220.0
    y_max = 1.5

    def x_of(delta):
        return x_left + (x_right - x_left) * math.log(delta / min(DELTAS_MIN)) / math.log(
            max(DELTAS_MIN) / min(DELTAS_MIN)
        )

    def y_of(value):
        return y_bottom - (y_bottom - y_top) * value / y_max

    pieces = [
        f'<svg class="fig" viewBox="0 0 {width} {height}" xmlns="http://www.w3.org/2000/svg" role="img" aria-label="Exact apparent switching rate falls as sampling interval grows; repeated estimates and model correction are shown with non-finite inversion frequencies.">',
        '<title>Endpoint sampling hides promoter switches</title>',
        '<desc>Exact p delta over delta curve, true rate reference, empirical middle-80-percent ranges from independent simulated records, and the fraction of records at or above one-half mismatch probability.</desc>',
    ]

    # Horizontal grid and inverse-minute tick labels.
    tick = 0.0
    while tick <= y_max + 1e-9:
        y = y_of(tick)
        pieces.append(
            f'<line x1="{x_left:.2f}" y1="{y:.2f}" x2="{x_right:.2f}" y2="{y:.2f}" stroke="var(--gridc)" stroke-width="1" />'
        )
        pieces.append(
            f'<text x="{x_left - 12:.2f}" y="{y + 5:.2f}" text-anchor="end" font-size="15" fill="var(--muted)" font-family="var(--sans)">{tick:.2f}</text>'
        )
        tick += 0.25

    # Axes and logarithmic sampling-interval ticks.
    pieces.extend([
        f'<line x1="{x_left:.2f}" y1="{y_top:.2f}" x2="{x_left:.2f}" y2="{y_bottom:.2f}" stroke="var(--line)" stroke-width="1.5" />',
        f'<line x1="{x_left:.2f}" y1="{y_bottom:.2f}" x2="{x_right:.2f}" y2="{y_bottom:.2f}" stroke="var(--line)" stroke-width="1.5" />',
    ])
    for delta in DELTAS_MIN:
        x = x_of(delta)
        pieces.append(
            f'<line x1="{x:.2f}" y1="{y_bottom:.2f}" x2="{x:.2f}" y2="{y_bottom + 6:.2f}" stroke="var(--line)" stroke-width="1.4" />'
        )
        pieces.append(
            f'<text x="{x:.2f}" y="{y_bottom + 25:.2f}" text-anchor="middle" font-size="15" fill="var(--ink)" font-family="var(--sans)">{delta:g}</text>'
        )

    # True-rate reference line.
    y_true = y_of(rate)
    pieces.append(
        f'<line x1="{x_left:.2f}" y1="{y_true:.2f}" x2="{x_right:.2f}" y2="{y_true:.2f}" stroke="var(--ink)" stroke-width="2" stroke-dasharray="7 6" />'
    )

    # The exact apparent-rate curve is calculated, not sketched.
    curve = []
    n_curve = 180
    log_min, log_max = math.log(min(DELTAS_MIN)), math.log(max(DELTAS_MIN))
    for index in range(n_curve + 1):
        fraction = index / n_curve
        delta = math.exp(log_min + fraction * (log_max - log_min))
        apparent = exact_mismatch_probability(rate, delta) / delta
        curve.append((x_of(delta), y_of(apparent)))
    path = " ".join(("M" if i == 0 else "L") + f"{x:.2f},{y:.2f}" for i, (x, y) in enumerate(curve))
    pieces.append(f'<path d="{path}" fill="none" stroke="var(--teal)" stroke-width="3.2" stroke-linecap="round" stroke-linejoin="round" />')

    # Repeat-simulation middle-80-percent ranges. Invalid inversions stay counted below.
    for summary in summaries:
        delta = summary["delta_min"]
        x = x_of(delta)
        naive = summary["naive_rate_range_p10_p50_p90"]
        x_naive = x - 9.0
        y1, ym, y2 = (y_of(naive[key]) for key in ("p10", "p50", "p90"))
        pieces.extend([
            f'<line x1="{x_naive:.2f}" y1="{y1:.2f}" x2="{x_naive:.2f}" y2="{y2:.2f}" stroke="var(--rose)" stroke-width="2.2" />',
            f'<line x1="{x_naive - 5:.2f}" y1="{y1:.2f}" x2="{x_naive + 5:.2f}" y2="{y1:.2f}" stroke="var(--rose)" stroke-width="2" />',
            f'<line x1="{x_naive - 5:.2f}" y1="{y2:.2f}" x2="{x_naive + 5:.2f}" y2="{y2:.2f}" stroke="var(--rose)" stroke-width="2" />',
            f'<path d="M{x_naive:.2f},{ym - 5:.2f} L{x_naive + 5:.2f},{ym:.2f} L{x_naive:.2f},{ym + 5:.2f} L{x_naive - 5:.2f},{ym:.2f} Z" fill="var(--rose)" stroke="var(--bg)" stroke-width="1" />',
        ])
        corrected = summary["corrected_finite_only_range_p10_p50_p90"]
        if corrected is not None:
            x_corrected = x + 9.0
            cy1, cym, cy2 = (y_of(corrected[key]) for key in ("p10", "p50", "p90"))
            pieces.extend([
                f'<line x1="{x_corrected:.2f}" y1="{cy1:.2f}" x2="{x_corrected:.2f}" y2="{cy2:.2f}" stroke="var(--plum)" stroke-width="2.2" />',
                f'<line x1="{x_corrected - 5:.2f}" y1="{cy1:.2f}" x2="{x_corrected + 5:.2f}" y2="{cy1:.2f}" stroke="var(--plum)" stroke-width="2" />',
                f'<line x1="{x_corrected - 5:.2f}" y1="{cy2:.2f}" x2="{x_corrected + 5:.2f}" y2="{cy2:.2f}" stroke="var(--plum)" stroke-width="2" />',
                f'<rect x="{x_corrected - 4.5:.2f}" y="{cym - 4.5:.2f}" width="9" height="9" rx="1.5" fill="var(--plum)" stroke="var(--bg)" stroke-width="1" />',
            ])
        finite_share = summary["p_hat_ge_half_fraction"] * 100.0
        pieces.append(
            f'<text x="{x:.2f}" y="{y_bottom + 74:.2f}" text-anchor="middle" font-size="14" fill="var(--muted)" font-family="var(--sans)">{finite_share:.1f}%</text>'

        f'<text x="80" y="{y_bottom + 74:.2f}" text-anchor="end" font-size="14" font-weight="600" fill="var(--ink)" font-family="var(--sans)">Non-finite</text>',
    ])

    # Compact two-row legend.
    legend_y1, legend_y2 = 328, 360
    pieces.extend([
        f'<line x1="82" y1="{legend_y1 - 5}" x2="116" y2="{legend_y1 - 5}" stroke="var(--teal)" stroke-width="3" />',
        f'<text x="124" y="{legend_y1}" font-size="14" fill="var(--ink)" font-family="var(--sans)">Exact apparent rate</text>',
        f'<line x1="315" y1="{legend_y1 - 5}" x2="315" y2="{legend_y1 + 5}" stroke="var(--rose)" stroke-width="2" /><path d="M315,{legend_y1 - 4} L320,{legend_y1 + 1} L315,{legend_y1 + 6} L310,{legend_y1 + 1} Z" fill="var(--rose)" />',
        f'<text x="328" y="{legend_y1}" font-size="14" fill="var(--ink)" font-family="var(--sans)">Naive endpoint rates</text>',
        f'<line x1="82" y1="{legend_y2 - 5}" x2="82" y2="{legend_y2 + 5}" stroke="var(--plum)" stroke-width="2" /><rect x="78" y="{legend_y2 - 4}" width="8" height="8" rx="1" fill="var(--plum)" />',

        '</svg>',
    ])
    return "\n".join(pieces)


def run_simulation():
    rng = random.Random(SEED)
    summaries = []
    replicate_records = []
    for delta in DELTAS_MIN:
        records = []
        for replicate in range(N_REPLICATES):
            record = simulate_dataset(K_PER_MIN, delta, N_INTERVALS, rng)
            record["replicate"] = replicate + 1
            record["delta_min"] = delta
            records.append(record)
        summaries.append(summarize_delta(K_PER_MIN, delta, records))
        replicate_records.extend(records)
    return {
        "assumptions": {
            "rate_per_min": K_PER_MIN,
            "deltas_min": list(DELTAS_MIN),
            "seed": SEED,
            "replicates_per_delta": N_REPLICATES,
            "sample_intervals_per_replicate": N_INTERVALS,
            "replicate_spread": "empirical 10th–90th percentiles (middle 80%), not confidence intervals",
            "trajectory": "exact continuous-time Markov jumps with exponential dwell times; only endpoint states recorded",
        },
        "summaries": summaries,
        "replicate_records": replicate_records,
        "svg": make_svg(summaries, K_PER_MIN),
    }


def print_report(data):
    assumptions = data["assumptions"]
    print("Illustrative symmetric two-state CTMC endpoint-sampling simulation")
    print(
        f"k={assumptions['rate_per_min']:g} min^-1; seed={assumptions['seed']}; "
        f"{assumptions['replicates_per_delta']} independent records × "
        f"{assumptions['sample_intervals_per_replicate']} intervals per Δ"
    )
    print("Middle-80% ranges are empirical repeat spread, not confidence intervals.")
    print("Δ(min)  exact pΔ   exact pΔ/Δ  p̂ range (10–90%)  naive/min (10–90%)  finite k̂/min (10–90%; n)  p̂≥1/2")
    for row in data["summaries"]:
        p = row["p_hat_range_p10_p50_p90"]
        n = row["naive_rate_range_p10_p50_p90"]
        c = row["corrected_finite_only_range_p10_p50_p90"]
        nfinite = row["corrected_finite_only_n"]
        nrep = row["n_replicates"]
        print(
            f"{row['delta_min']:>5g}   {row['exact_p_delta']:.6f}   "
            f"{row['exact_naive_rate_per_min']:.6f}   "
            f"{_fmt(p['p10'], 4)}–{_fmt(p['p90'], 4)}       "
            f"{_fmt(n['p10'])}–{_fmt(n['p90'])}             "
            f"{_fmt(c['p10'])}–{_fmt(c['p90'])}; {nfinite}/{nrep}        "
            f"{100 * row['p_hat_ge_half_fraction']:.1f}%"
        )
        counts = row["status_counts"]
        print(
            f"      inversion statuses: finite={counts['finite']}, "
            f"boundary p̂=1/2→∞={counts['boundary_infinite']}, "
            f"above 1/2=no finite inverse={counts['above_half_no_finite_inverse']}"
        )


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--json", action="store_true", help="emit full structured results, including every replicate")
    args = parser.parse_args()
    data = run_simulation()
    if args.json:
        print(json.dumps(data, ensure_ascii=False, allow_nan=False, separators=(",", ":")))
    else:
        print_report(data)
        print(f"Generated inline SVG: {len(data['svg'].encode('utf-8'))} bytes")


if __name__ == "__main__":
    main()

What I would argue against / where this breaks

This inverse assumes two observed states, equal constant hazards, known fixed Δ, and error-free endpoints. Hidden states, unequal rates, missed frames, misclassification or timing jitter change the transition law. At large Δ, inversion is unstable; p^Δ≥12 signals a sample boundary or model mismatch, not an infinite biological rate.
